- [ ] **Step 3: The codemod**

```js
// tools/lint/migrate-directives.mjs
//
// Copyright (C) 2026 Posit Software, PBC
//
// deno-lint-ignore directives -> oxlint directives.
// Usage: node tools/lint/migrate-directives.mjs <dir>...
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

// null: delete (rule off, or not in oxlint and not in deno's recommended set)
const kRuleMap = {
  "no-explicit-any": "typescript/no-explicit-any",
  "no-this-alias": "typescript/no-this-alias",
  "no-control-regex": "eslint/no-control-regex",
  "no-cond-assign": "eslint/no-cond-assign",
  "prefer-const": "eslint/prefer-const",
  "no-prototype-builtins": "eslint/no-prototype-builtins",
  "no-empty": "eslint/no-empty",
  "no-unused-vars": "eslint/no-unused-vars",
  "no-fallthrough": "eslint/no-fallthrough",
  "no-case-declarations": "eslint/no-case-declarations",
  "no-window-prefix": "eslint/no-restricted-globals",
  "no-window": "eslint/no-restricted-globals",
  "require-await": null,
  "no-inferrable-types": null,
  "camelcase": null,
};

function mapRules(rules) {
  const out = [];
  for (const r of rules) {
    if (!(r in kRuleMap)) {
      throw new Error(`unknown deno-lint rule "${r}"; add it to kRuleMap in tools/lint/migrate-directives.mjs`);
    }
    if (kRuleMap[r] && !out.includes(kRuleMap[r])) out.push(kRuleMap[r]);
  }
  return out;
}

export function migrateDirectives(source) {
  const out = [];
  for (const line of source.split("\n")) {
    let m = line.match(/^(\s*)\/\/\s*deno-lint-ignore-file(?:\s+(.*?))?\s*$/);
    if (m) {
      const rules = m[2] ? mapRules(m[2].split(/\s+/)) : [];
      if (m[2] && rules.length === 0) continue;
      out.push(`${m[1]}/* oxlint-disable${rules.length ? " " + rules.join(", ") : ""} */`);
      continue;
    }
    m = line.match(/^(\s*)\/\/\s*deno-lint-ignore\s+(.*?)\s*$/);
    if (m) {
      const rules = mapRules(m[2].split(/\s+/));
      if (rules.length > 0) out.push(`${m[1]}// oxlint-disable-next-line ${rules.join(", ")}`);
      continue;
    }
    out.push(line);
  }
  return out.join("\n");
}

function walk(dir, files) {
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) {
      if (e.name !== "node_modules" && e.name !== ".git") walk(p, files);
    } else if (/\.(ts|mts|js|mjs)$/.test(e.name)) {
      files.push(p);
    }
  }
  return files;
}

if (process.argv[1] && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  let changed = 0;
  for (const dir of process.argv.slice(2)) {
    for (const file of walk(dir, [])) {
      const before = fs.readFileSync(file, "utf8");
      if (!before.includes("deno-lint-ignore")) continue;
      const after = migrateDirectives(before);
      if (after !== before) {
        fs.writeFileSync(file, after);
        changed++;
      }
    }
  }
  console.log(`migrated directives in ${changed} files`);
}
```
